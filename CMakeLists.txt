cmake_minimum_required(VERSION 3.24)
project(InputOutput VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "InputOutput targets Windows 11.")
endif()
add_executable(InputOutput WIN32
    src/main.cpp src/app.cpp src/ui.cpp src/audio.cpp src/display.cpp src/storage.cpp src/tests.cpp
    assets/app.rc)
target_compile_features(InputOutput PRIVATE cxx_std_20)
target_compile_definitions(InputOutput PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_include_directories(InputOutput PRIVATE assets)
target_link_libraries(InputOutput PRIVATE user32 shell32 ole32 uuid propsys advapi32 comctl32 dwmapi gdi32 psapi uxtheme)
if(MSVC)
    set_property(TARGET InputOutput PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(InputOutput PRIVATE /W4 /utf-8 /permissive-)
else()
    target_compile_options(InputOutput PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    target_link_options(InputOutput PRIVATE -static -municode)
endif()
enable_testing()
add_test(NAME InputOutputSelfTests COMMAND InputOutput --self-test --data-dir "${CMAKE_BINARY_DIR}/self-test")
